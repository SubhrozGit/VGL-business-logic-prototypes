ROUTINE AuthoriseJob( VALUE curjob)

DECLARE DBStatus
DECLARE doTransaction
DECALRE DBCheck
DECLARE curmsg

DBStatus = select JobHeader.Job_name FOR UPDATE WHERE job_name = curjob
IF IsValue(DBStatus) THEN
   IF (DBStatus <> LOCKED) THEN
       doTransaction = NOT(transaction_is_write())
            
           IF (doTransaction) THEN
      
                      START WRITE TRANSACTION "Authorise Job"
                      ASSIGN job_header.authoriser = "Subhradeep"
                      WRITE jo_header
                      ENDIF
                      COMMIT
                      curmsg : "Job": TRIM(curjob):"successfully authorized"
                      WRITE log_file(logfileName,"authoriser", curmsg)
              
            IF(transaction_is_write()) THEN
                      
                      ROLLBACK
                      curmsg : "Job": TRIM(curjob):"cannot be authorized at the moment" 
                      WRITE log_file(logfileName,"cannot authorize", curmsg)  
            ENDIF

            ELSEIF(DBStatus == LOCKED) THEN
                      ROLLBACK
                      curmsg : "Job": TRIM(curjob):"cannot be authorized"
                      WRITE log_file(logfileName,"LOCK", curmsg)
            ENDIF
ENDIF
ENDROUTINE
